add_library(sqlite STATIC sqlite3.c)
add_library(SQLite::SQLite3 ALIAS sqlite)
target_include_directories(sqlite PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1)
target_link_libraries(sqlite PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(sqlite PRIVATE m)
endif()
set_target_properties(sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
