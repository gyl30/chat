add_library(chat_sqlite STATIC sqlite3.c)
add_library(SQLite::SQLite3 ALIAS chat_sqlite)
target_include_directories(chat_sqlite PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(chat_sqlite PRIVATE SQLITE_THREADSAFE=1)
target_link_libraries(chat_sqlite PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(chat_sqlite PRIVATE m)
endif()
set_target_properties(chat_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
